```typescript
/**
 * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
 *
 * 
 * @param s string字符串 
 * @param n int整型 
 * @return string字符串
 */
export function trans(s: string, n: number): string {
     // 1. 把字符串分割为字符
    let letterArr = s.split('');

    // 2. 字母反转
    letterArr = letterArr.map(le => {
        if (/^[a-z]$/.test(le)) {
            return le.toUpperCase();
        } else if (/^[A-Z]$/.test(le)) {
            return le.toLowerCase();
        } 
        return le;
    })

    // 3. 把反转字母拼接回来
    s = letterArr.join('')

    // 4. 把单词反序
    s = s.split(' ').reverse().join(' ');

    return s
}
```

### 最长公共前缀

##### 题解

```typescript
/**
 * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
 *
 * 
 * @param strs string字符串一维数组 
 * @return string字符串
 */
export function longestCommonPrefix(strs: string[]): string {
    // 1. 处理边界情况
    if (!strs.length) return '';

    // 2. 如果只有一个字符串，说明第一个字符串就是公共子串
    if (strs.length === 1) return strs[0];

    // 3. 以第一个字符串作为标杆查找公共子串
    const firstStr = strs[0];
    for (let i = 0; i < firstStr.length; i += 1) {
        const char = firstStr[i];

        // 第一个字符串是标杆，不需要匹配了
        for (let j = 1; j < strs.length; j +=1 ) {
            
            // 1. 如果i位置的字符不相同，说明匹配完了
            if (i === strs[j].length || char !== strs[j][i]) {
                return firstStr.substring(0, i);
            }
        }
    }

    // 4. 如果全都执行完了，说明全部匹配上了
    return firstStr;
}
```
