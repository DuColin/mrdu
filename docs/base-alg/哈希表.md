```typescript
/**
 * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
 *
 * 
 * @param numbers int整型一维数组 
 * @param target int整型 
 * @return int整型一维数组
 */
export function twoSum(numbers: number[], target: number): number[] {
   const map = new Map() // 存储的是<value, key>映射

   for (let i = 0; i < numbers.length; i += 1) {
        const curVal = numbers[i];
        const subVal = target - curVal;

        if (map.has(subVal)) {
            return [map.get(subVal), i + 1]
        }

        map.set(curVal, i + 1);
   }
}
```

```typescript
export function MoreThanHalfNum_Solution(numbers: number[]): number {
    // 1. 记录每个值出现的次数
    const map = new Map<number, number>();

    const half = numbers.length / 2;
    
    for (let i = 0; i < numbers.length; i += 1) {
        const value = numbers[i];

        // 2. 获取当前值存在的次数
        let count = map.get(value) ?? 0;

        // 3. 增加本轮次数
        count += 1;
        map.set(value, count);

        // 4. 比较次数
        if (count > half) {
            return value;
        }
    }

```

```typescript
/**
 * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
 *
 * 
 * @param nums int整型一维数组 
 * @return int整型一维数组
 */
export function FindNumsAppearOnce(nums: number[]): number[] {

    // write code here
    const countMap = new Map();

    for (let i = 0; i < nums.length; i += 1) {
        const value = nums[i];

        const count = countMap.get(value) ?? 0;
        countMap.set(value, count + 1);
    }

    

    const filterArray = Array.from(countMap.entries()).filter(([value, count]) => {
        return count === 1;
    }).map(([value]) => {
        return value;
    }).sort((a, b) => {
         return a - b
    });

    return filterArray;
}
```


### 4.  缺失的第一个正整数

##### 描述

##### 题解

**题目解读**

- **算法选择**：题目给出的时间、空间复杂度啊要求，意味着我们不能使用任何新的数组，必须要在原数组 nums 上做文章。
- **隐藏条件**：负数（以及 0、大于数组长度 $n$ 的大数）在题目中是无效的，是一个干扰项，它们统统属于“没票的黑户”或者“拿了假票的人”。所以我们解题时不需要考虑它们的位置。

**代码实现**

```typescript 
/**
 * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
 *
 * 
 * @param nums int整型一维数组 
 * @return int整型
 */
export function minNumberDisappeared(nums: number[]): number {
    
    const n = nums.length;

    // 1. 数据重排。让数字做到正确的位置上：1-> 0, 2->1, 3-> 2 .... n->n-1
    for (let i = 0; i < nums.length; i += 1) {

        /**
         * 这里需要不断处理 nus[i]位置的数字，全都把它放到正确的位置上
         * val > 0 : 如果数字小于零则不用管它
         * val <= n : 如果数字超出了 n 的范围也不应该管它 
         * val !== nums[targetIndex]: 如果数字在正确的位置上，也不需要管它
         */
        while (nums[i] > 0 && nums[i] <= n && nums[i] !== nums[nums[i] -1]) {
            // 当前数字应该坐的正确的位置
            const targetIndex = nums[i] - 1;
            // 此时位置上做的数字是谁
            const targetIndexOldVal = nums[targetIndex]
            // 交换两个位置的值
            nums[targetIndex] = nums[i];
            nums[i] = targetIndexOldVal;
        }
    }

    // 2. 查找重排后的数字，哪个没有做到自己正确的位置上
    for (let i = 0; i < n; i += 1) {
        if (nums[i] !== i + 1) {
            return i + 1;
        }
    }

    // 3. 如果所有的数字都坐在正确的位置上，则说明缺失的数字是 n + 1
    return n + 1
}
```
### 5.  三数之和

##### 描述
##### 题解

```typescript
/**
 * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
 *
 * 
 * @param num int整型一维数组 
 * @return int整型二维数组
 */
export function threeSum(num: number[]): number[][] {

    // 1. 首先对数组进行排序，方便答案去重和查找
    num = num.sort((a, b) => a - b);

    // 2. 查找所有满足三数之和等于0的数据
    const n = num.length;
    const results = []
    for (let i = 0; i < n - 2; i += 1) {
        // 2.1 如果第一个数字是大于0，由于是升序排列的，后续所有数字都不可能相加等于0
        if (num[i] > 0) break;

        // 2.2 如果这个数字和上一个数字相同，则说明已经找过了，不在重新查找
        if (i > 0 && num[i] === num[i -1]) continue;

        // 2.3 使用双指针，查找剩余两个符合条件的数字。
        let left = i + 1;
        let right = n - 1;

        while(left < right) {
            const sum = num[i] + num[left] + num[right];

            if (sum === 0) {
                results.push([num[i], num[left], num[right]]);

                while(left < right && num[left] == num[left + 1]) left += 1;
                while(left < right && num[right] === num[right -1]) right -=1;

                left += 1;
                right -= 1;
            } else if (sum > 0) {
                // 如果sum > 0 ，说明剩余的数字大了
                right -= 1;
            } else if (sum < 0) {
                // 如果sum > 0 ，说明剩余的数字笑了
                left += 1;
            }
        }
    }

    return results;
}
```
