# 动态规划



## 题目

### 1. 斐波那契数列

##### 题解

**代码实现**

```ts
/**
 * @param n int整型 
 * @return int整型
 */
export function Fibonacci(n: number): number {
    // write code here

    // 1. 处理边界
    if (n === 1 || n === 2) return 1;

    // 2. 计算 n
    let pre1 = 1;
    let pre2 = 1;
    let current = 0;

    for (let i = 3; i <= n; i += 1) {
        current = pre1 + pre2;
        pre1 = pre2;
        pre2 = current;   
    }

    return current;
}
```
